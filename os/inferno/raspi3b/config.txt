# config.txt - Raspberry Pi 3 Model B, bare-metal Inferno kernel.
# The VideoCore firmware reads this, loads the ARM kernel image into RAM
# at 0x80000, and releases core 0 into it (at EL2 on a 64-bit kernel).

arm_64bit=1
kernel=kernel8.img

# Console: this kernel drives the AUX mini UART (UART1, 0x3f215040) on
# GPIO 14/15 (header pins 8 TX, 10 RX), 115200 8N1.  That is the Pi 3's
# default wiring once the UART is enabled, so no uart overlay is named:
# the PL011 stays on the Bluetooth part.  (disable-bt and miniuart-bt
# would each move the PL011 back onto GPIO 14/15, away from this driver.)
enable_uart=1

# The mini UART's baud divisor counts the VideoCore core clock, so pin it
# (enable_uart=1 implies this on a Pi 3; say it anyway) and keep the
# firmware from scaling it underneath the console.
core_freq=250
