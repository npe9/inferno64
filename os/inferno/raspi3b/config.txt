# config.txt - Raspberry Pi 3 Model B, bare-metal Inferno kernel.
# The VideoCore firmware reads this, loads the ARM kernel image into RAM
# at 0x80000, and releases core 0 into it (at EL2 on a 64-bit kernel).

arm_64bit=1
kernel=kernel8.img

# Console, 115200 8N1 on GPIO 14/15 (header pins 8/10).
#
# On a Pi 3 the PL011 belongs to Bluetooth and the header gets the mini
# UART.  The kernel takes whichever the firmware wired -- it reads GPIO
# 14's function at boot -- so this file chooses, and choosing the
# default leaves the PL011 free for Bluetooth later, which is the only
# UART the BCM43438 is on.
#
# dtoverlay=disable-bt swaps them, giving the console the PL011 and
# giving up Bluetooth.  It is left out on purpose, and not only for
# Bluetooth's sake: an overlay is a file on this card, and when the
# board's own file server damaged the FAT the overlay vanished, the
# firmware fell back to the default, and every kernel print went out
# the Bluetooth pins.  A console that depends on a file can disappear.
enable_uart=1

# Pin the core clock so the UART baud is stable (and don't let the firmware
# downclock when idle, which would move the baud divisor under us).
core_freq=250

# JTAG on the header, GPIO 22-27 in ALT4.
#
# Two days of bring-up went like this: build a kernel, write 1.9MB to
# the card through a filesystem server that is itself one of the things
# under test, read it all back to verify, reboot, and hope.  Three
# minutes when it worked; twice it silently installed the wrong bytes,
# and twice more the card came out and went into a reader.  A debugger
# that loads straight into RAM at 0x80000 never touches the card at
# all, so that whole class of failure stops existing.
#
# It is also the instrument this port keeps wanting and not having.
# The SD DMA stall was diagnosed by adding a print, rebuilding,
# installing and rebooting once per question; with a halt-and-inspect
# debugger the controller's registers at the moment of the stall are a
# read, not a release cycle.
#
# The pins, which are the 40-pin header's own:
#
#	GPIO22  TRST  pin 15
#	GPIO23  RTCK  pin 16
#	GPIO24  TDO   pin 18
#	GPIO25  TCK   pin 22
#	GPIO26  TDI   pin 37
#	GPIO27  TMS   pin 13
#
# This costs nothing when no adapter is attached: the firmware sets the
# alt function and the pins sit idle.  It does take GPIO 22-27 away
# from anything else that wants them, and nothing on this board does.
enable_jtag_gpio=1
