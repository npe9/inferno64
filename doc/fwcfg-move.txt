The fw_cfg reader has to be where every configuration links it.

Verified in a detached worktree at multicore's 225c058c: with this
applied, both os/inferno/virtarm64 configurations build -- ivirtarm64
1848059 bytes and ivirtarm64cloud 2205080 -- where before the cloud one
failed at "fwcfgbootargs: undefined: fwcfgfile", because bootargs.c
reads the boot options out of fw_cfg and the reader lived in
screenarm.c, which virtarm64cloud does not link (probing ramfb at a
fixed address kills a Compute Engine t2a).

It is recorded here rather than applied because the file it moves code
out of is the other branch's: their screenarm.c made fwcfgdma
non-static and gave it a static descriptor, since the options are read
before the allocator exists, and that version is the one that survives.
Apply this on their line, or fold it into the merge resolution.

See doc/merge-parallelvm-cloud.ms.

+int
+fwcfgdma(u32int control, void *buf, u32int len)
+{
+	/*
+	 * A static descriptor, not a malloc'd one.  bootargsinit() reads
+	 * the boot options through here, and it runs before the allocator
+	 * exists -- it has to, because the loader leaves its block in
+	 * memory the kernel is about to allocate over.  malloc() returned
+	 * nil there and this returned -1, so the options silently did not
+	 * arrive and every *parallelvm, *soak and *secsched on a -kernel
+	 * boot line was ignored.
+	 *
+	 * Safe because fw_cfg is read at boot, by one processor: the
+	 * options here, and the framebuffer below before any other
+	 * processor is started.
+	 */
+	static FwDmaDesc dmabuf;
+	FwDmaDesc *dma;
+	u64int dmap;
+
+	dma = &dmabuf;
+	memset(dma, 0, sizeof(*dma));
+	dma->control = be32(control);
+	dma->length = be32(len);
+	dma->address = be64((u64int)(uintptr)buf);
+	coherence();
+	dmap = (u64int)(uintptr)dma;
+	*(volatile u32int*)FwCfgDmaReg = be32((u32int)(dmap>>32));
+	*(volatile u32int*)(FwCfgDmaReg+4) = be32((u32int)dmap);
+	while(be32(dma->control) & ~(u32int)1)
+		;
+	if(be32(dma->control) & 1)
+		return -1;
+	return 0;
+}

+ * (measured: fw_cfg says 0 bytes).  Returns the selector, and the item's

+int
+fwcfgfile(char *name, u32int *size)
+{
+	u32int count, i;
+	uchar ent[64];
+	u16int sel;
+	int n;
+
+	n = strlen(name);
+	if(fwcfgdma(((u32int)FwCfgFileDir<<16) | FwCfgDmaSelect | FwCfgDmaRead,
+	    &count, 4) < 0)
+		return -1;
+	count = be32(count);
+	if(count > 512)
+		count = 512;
+	for(i = 0; i < count; i++){
+		if(fwcfgdma(FwCfgDmaRead, ent, 64) < 0)
+			return -1;
+		sel = ((u16int)ent[4]<<8) | ent[5];
+		if(strncmp((char*)(ent+8), name, n) == 0 && ent[8+n] == 0){
+			if(size != nil)
+				*size = ((u32int)ent[0]<<24) | ((u32int)ent[1]<<16)
+					| ((u32int)ent[2]<<8) | ent[3];
+			return (int)sel;
+		}
+	}
+	return -1;
+}
